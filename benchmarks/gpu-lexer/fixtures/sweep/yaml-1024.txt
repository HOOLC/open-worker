name: CI

on:
  push:
    branches:
      - main
  pull_request:

concurrency:
  group: ci-${{ github.ref }}
  cancel-in-progress: true

permissions:
  contents: read

jobs:
  test:
    name: Build and test
    runs-on: ubuntu-latest
    timeout-minutes: 15

    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Set up kache build cache
        uses: kunobi-ninja/kache-action@v1
        with:
          s3-bucket: zork-kache
          s3-region: auto
          s3-endpoint: https://00000000000000000000000000000000.r2.cloudflarestorage.com
          s3-access-key-id: ${{ secrets.KACHE_S3_ACCESS_KEY }}
          s3-secret-access-key: ${{ secrets.KACHE_S3_SECRET_KEY }}

      - name: Set up pnpm
        uses: pnpm/action-setup@v4
        with:
          version: 10.33.0

      - name: Set up Node.js
        uses: actions/setup-node@v4
        with:
          node-version: 22.15
          cache: pnpm

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      -
